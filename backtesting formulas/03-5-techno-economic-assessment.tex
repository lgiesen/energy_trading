\section{Techno Economic Assessment}\label{ch:methodology:techno-economic-assessment}
\subsection{Revenue Components}
Ex-post revenues are computed from settled grid-side quantities.

Day-ahead (DA) revenue and cost:
\begin{equation}\label{eq:ta-rev-da}
    Rev^{DA}_t = E^{DA}_{\mathrm{sell},t}\,\lambda^{DA}_t
\end{equation}
\begin{equation}\label{eq:ta-cost-da}
    Cost^{DA}_t = E^{DA}_{\mathrm{buy},t}\,\lambda^{DA}_t
\end{equation}

Intraday (ID) rescue revenue and cost:
\begin{equation}\label{eq:ta-id-prices}
    p^{ID}_{\mathrm{buy},t}=\min\!\left(3000,\lambda^{DA}_t+30\right),
    \qquad
    p^{ID}_{\mathrm{sell},t}=\max\!\left(-500,\lambda^{DA}_t-30\right)
\end{equation}
\begin{equation}\label{eq:ta-rev-id}
    Rev^{ID}_t = E^{ID}_{\mathrm{sell},t}\,p^{ID}_{\mathrm{sell},t}
\end{equation}
\begin{equation}\label{eq:ta-cost-id}
    Cost^{ID}_t = E^{ID}_{\mathrm{buy},t}\,p^{ID}_{\mathrm{buy},t}
\end{equation}

aFRR capacity and activation revenue:
\begin{equation}\label{eq:ta-rev-cap}
    Rev^{cap}_t = R^{+}_{t}\,C^{cap,+}_{t}\,\Delta t + R^{-}_{t}\,C^{cap,-}_{t}\,\Delta t
\end{equation}
\begin{equation}\label{eq:ta-rev-act}
    Rev^{act}_t = E^{act,+}_{\mathrm{del},t}\,\lambda^{act,+}_{t} + E^{act,-}_{\mathrm{del},t}\,\lambda^{act,-}_{t}
\end{equation}
\subsection{Cost, Degradation and Penalty Components}
Transaction cost is applied to total grid-side throughput:
\begin{equation}\label{eq:ta-cost-tx}
    C^{tx}_t
    =
    c_{tx}\Big(
    E^{DA}_{\mathrm{buy},t}+E^{DA}_{\mathrm{sell},t}
    +E^{ID}_{\mathrm{buy},t}+E^{ID}_{\mathrm{sell},t}
    +E^{act,+}_{\mathrm{del},t}+E^{act,-}_{\mathrm{del},t}
    \Big)
\end{equation}

Degradation is charged on internal battery throughput:
\begin{equation}\label{eq:ta-cost-deg}
    C^{deg}_t
    =
    c_{deg}\Big(
    E^{int,DA}_{ch,t}+E^{int,DA}_{dis,t}
    +E^{int,ID}_{ch,t}+E^{int,ID}_{dis,t}
    +E^{int,act,+}_{t}+E^{int,act,-}_{t}
    \Big)
\end{equation}

Capacity and activation non-delivery are penalized using a proxy based on missed quantities and basis prices:
\begin{equation}\label{eq:ta-missed-cap}
    M^{+}_{t}=\max\!\left(0, R^{+}_{t}-R^{+}_{\max,t}\right),
    \qquad
    M^{-}_{t}=\max\!\left(0, R^{-}_{t}-R^{-}_{\max,t}\right)
\end{equation}
\begin{equation}\label{eq:ta-missed-act}
    E^{act,+}_{miss,t}=\max\!\left(0,E^{act,+}_{req,t}-E^{act,+}_{del,t}\right),
    \qquad
    E^{act,-}_{miss,t}=\max\!\left(0,E^{act,-}_{req,t}-E^{act,-}_{del,t}\right)
\end{equation}
\begin{equation}\label{eq:ta-pen-total}
    C^{pen}_t = C^{pen,act}_t + C^{pen,cap}_t
\end{equation}

Equivalent Full Cycles (EFC) are reported from discharge-side internal throughput:
\begin{equation}\label{eq:ta-efc}
    \mathrm{EFC}
    =
    \frac{
    \sum_t\left(E^{DA}_{\mathrm{sell},t}+E^{ID}_{\mathrm{sell},t}+E^{act,+}_{\mathrm{del},t}\right)/\eta_{\text{out}}
    }{E_{\text{cap}}}
\end{equation}
\subsection{Baseline Strategies}
Baseline strategy variants are evaluated with identical settlement equations (Sections~\ref{eq:ta-rev-da}--\ref{eq:ta-pen-total}):
\begin{itemize}
    \item \texttt{da\_only}: reserve decisions constrained to zero.
    \item \texttt{afrr\_only}: DA charge/discharge decisions constrained to zero.
    \item \texttt{naive}: simplified forecast-driven dispatch benchmark.
    \item \texttt{oracle}: perfect-foresight upper bound using realized series in planning.
\end{itemize}
No separate settlement formulas are used for baselines.
\subsection{Economic Settlement and Performance Metrics / Performance Indicators}
Hourly net cashflow (excluding non-cash degradation):
\begin{equation}\label{eq:ta-cashflow}
    CF^{net}_t
    =
    Rev^{DA}_t-Cost^{DA}_t
    +Rev^{cap}_t+Rev^{act}_t
    +Rev^{ID}_t-Cost^{ID}_t
    -C^{tx}_t-C^{pen}_t
\end{equation}

Hourly accounting PnL:
\begin{equation}\label{eq:ta-pnl-hour}
    \pi_t = CF^{net}_t - C^{deg}_t
\end{equation}

Terminal inventory valuation (implementation convention):
\begin{equation}\label{eq:ta-terminal}
    V_{term}=(SoC_T-SoC_0)\,\eta_{\text{out}}\,\lambda^{term}_{DA}
\end{equation}

Final realized total:
\begin{equation}\label{eq:ta-pnl-total}
    \Pi^{real}=\sum_t \pi_t + V_{term}
\end{equation}

Derived indicators include:
\begin{itemize}
    \item total penalty exposure $\sum_t C^{pen}_t$,
    \item total degradation $\sum_t C^{deg}_t$,
    \item EFC (Eq.~\ref{eq:ta-efc}),
    \item profit per EFC: $\Pi^{real}/\mathrm{EFC}$ (when $\mathrm{EFC}>0$),
    \item average daily profit over the simulated period.
\end{itemize}

Comparative indicators are computed consistently across model, quantile, and baseline runs, including realized-versus-oracle capture ratios.
